function generar(){ flushInputs(); const fecha = document.getElementById('fecha').value; const turno = document.getElementById('turno').value; let out = `Inventario – ${fecha} – Turno ${turno}\n\n`; out += `🔴 ROJO — REPOSICIÓN\n🟡 AMARILLO — BAJO / VIGILAR\n🟢 VERDE — STOCK ÓPTIMO\n\n`; const STATUS_EMOJI = {verde:'🟢', amarillo:'🟡', rojo:'🔴'}; const mermaLineas = []; const rojos = [], amarillos = [], pedidos = []; categorias.forEach(cat => { const lineas = cat.items.map(item => { const d = datos[key(cat.nombre,item)]; if(!d) return null; // Resumen: se junta aunque no tenga cantidad escrita const cant = d.cantidad !== '' ? d.cantidad : 's/d'; const linea = `- ${item} (${cat.nombre}): ${cant}`; if(d.status === 'rojo') rojos.push(linea + (d.pedir ? ' → PEDIR' : '')); if(d.status === 'amarillo') amarillos.push(linea + (d.pedir ? ' → PEDIR' : '')); if(d.pedir) pedidos.push(linea + ' ' + STATUS_EMOJI[d.status]); // Lista completa: solo los que tienen cantidad if(d.cantidad === '') return null; const mermaTag = (d.merma && d.merma !== '0') ? ` ⚠️merma:${d.merma}` : ''; if(d.merma && d.merma !== '0') mermaLineas.push(`- ${item} (${cat.nombre}): ${d.merma}`); return `- ${item}-${d.cantidad}${STATUS_EMOJI[d.status]}${mermaTag}${d.pedir?' PEDIR':''}`; }).filter(Boolean); if(lineas.length > 0) out += `${cat.nombre}\n\n${lineas.join('\n')}\n\n`; }); // ── RESUMEN DE ALERTAS ── out += `═════════════════════\n📌 RESUMEN DE ALERTAS\n\n`; out += `🔴 ROJO — REPOSICIÓN (${rojos.length})\n`; out += rojos.length ? rojos.join('\n') + '\n\n' : '- Ninguno\n\n'; out += `🟡 AMARILLO — BAJO / VIGILAR (${amarillos.length})\n`; out += amarillos.length ? amarillos.join('\n') + '\n\n' : '- Ninguno\n\n'; out += `🛒 PARA PEDIR (${pedidos.length})\n`; out += pedidos.length ? pedidos.join('\n') + '\n\n' : '- Nada marcado\n\n'; if(mermaLineas.length > 0){ out += `─────────────────────\n⚠️ RESUMEN DE MERMA\n\n${mermaLineas.join('\n')}\n`; } const pre = document.getElementById('result-pre'); const empty = document.getElementById('result-empty'); pre.textContent = out.trim(); pre.style.display = 'block'; empty.style.display = 'none'; switchTab('resultado'); }
